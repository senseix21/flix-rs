cmake_minimum_required(VERSION 3.16)
project(flix_rust_gazebo LANGUAGES CXX)
find_package(gazebo 11 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(flix_rust SHARED plugin.cpp)
target_include_directories(flix_rust PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(flix_rust PRIVATE ${GAZEBO_LIBRARY_DIRS})
set(FLIX_RUST_LIBRARY "${ROOT}/target/release/libflight_gazebo.a" CACHE FILEPATH "Rust static archive for this host platform")
target_link_libraries(flix_rust PRIVATE "${FLIX_RUST_LIBRARY}" ${GAZEBO_LIBRARIES} pthread dl m)
# Legacy comparison: build the pinned ../flix/gazebo plugin separately and select its SDF.
add_executable(flix-world-reset world-reset.cpp)
target_include_directories(flix-world-reset PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(flix-world-reset PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(flix-world-reset PRIVATE ${GAZEBO_LIBRARIES} pthread tbb)
